## 1. `src/features/system/user/components/user-search-form.tsx` (new file)

```tsx
import { Button, DatePicker, Form, Input, Space } from 'antd'
import type { Dayjs } from 'dayjs'
import { useTranslation } from 'react-i18next'
import { DeptTreeSelect } from '@/shared/components/dept-tree-select'
import { DictSelect } from '@/shared/components/dict-select'
import { USER_DICT_TYPES } from '@/features/system/user/constants'
import type { UserFilters } from '@/features/system/user/types'

/**
 * User search form — 5 inline fields per T2.0-D11.
 *
 * Form value shape (`SearchFormValues`) is intentionally INTERNAL to this
 * component — uses Dayjs for the RangePicker. The form converts to domain
 * shape (`UserFilters` with ISO Instant tuple) on submit, so the parent
 * never deals with Dayjs.
 *
 * Empty strings stripped to undefined to avoid sending `username=` etc.
 * (BE treats empty string as "match empty", not "no filter").
 */

interface SearchFormValues {
  username?: string
  mobile?: string
  status?: number
  deptId?: number
  createTime?: [Dayjs, Dayjs]
}

interface UserSearchFormProps {
  onSearch: (filters: UserFilters) => void
  onReset: () => void
  loading?: boolean
}

export function UserSearchForm({ onSearch, onReset, loading }: UserSearchFormProps) {
  const { t } = useTranslation()
  const [form] = Form.useForm<SearchFormValues>()

  const handleFinish = (values: SearchFormValues) => {
    const filters: UserFilters = {
      username: values.username?.trim() || undefined,
      mobile: values.mobile?.trim() || undefined,
      status: values.status,
      deptId: values.deptId,
      createTime: values.createTime
        ? [
            values.createTime[0].startOf('day').toISOString(),
            values.createTime[1].endOf('day').toISOString(),
          ]
        : undefined,
    }
    onSearch(filters)
  }

  const handleReset = () => {
    form.resetFields()
    onReset()
  }

  return (
    <Form
      form={form}
      layout="inline"
      onFinish={handleFinish}
      style={{ marginBottom: 16, rowGap: 16, flexWrap: 'wrap' }}
    >
      <Form.Item name="username" label={t('systemUser.search.username')}>
        <Input placeholder={t('systemUser.search.username')} allowClear style={{ width: 180 }} />
      </Form.Item>

      <Form.Item name="mobile" label={t('systemUser.search.mobile')}>
        <Input placeholder={t('systemUser.search.mobile')} allowClear style={{ width: 180 }} />
      </Form.Item>

      <Form.Item
        name="status"
        label={t('systemUser.search.status')}
        normalize={v => (v == null || v === '' ? undefined : Number(v))}
      >
        <DictSelect
          dictType={USER_DICT_TYPES.status}
          allowClear
          placeholder={t('systemUser.search.status')}
          style={{ width: 140 }}
        />
      </Form.Item>

      <Form.Item name="deptId" label={t('systemUser.search.deptId')}>
        <DeptTreeSelect
          allowClear
          placeholder={t('systemUser.search.deptId')}
          style={{ width: 220 }}
        />
      </Form.Item>

      <Form.Item name="createTime" label={t('systemUser.search.createTime')}>
        <DatePicker.RangePicker style={{ width: 260 }} />
      </Form.Item>

      <Form.Item>
        <Space>
          <Button type="primary" htmlType="submit" loading={loading}>
            {t('common.search')}
          </Button>
          <Button onClick={handleReset}>{t('common.reset')}</Button>
        </Space>
      </Form.Item>
    </Form>
  )
}
```

### Design notes

- **`normalize` for status** (A2-TD2 workaround): DictSelect emits string values from BE dict. `normalize` converts to number at form layer so `filters.status: number | undefined` is correct shape at submit time.
- **`.trim()` on string fields**: defensive — user pastes whitespace, BE LIKE query treats as substring. Strip leading/trailing.
- **No `onValuesChange`** — search only fires on explicit Submit (per T2.0-D11). Reset clears form + tells parent to clear filters.
- **`flexWrap: 'wrap'` + `rowGap: 16`** — antd inline form doesn't auto-wrap gracefully on narrow screens; adding flex-wrap rules covers it.

## 2. `src/features/system/user/components/user-list-page.tsx` (full replacement)

```tsx
import { useMutation, useQueryClient } from '@tanstack/react-query'
import { App, Button, Card, Space, Switch, Table, type TableColumnsType } from 'antd'
import { useState } from 'react'
import { useTranslation } from 'react-i18next'

import { selectUser } from '@/app/slices/auth-slice'
import { useAppSelector } from '@/app/store'
import { HasPermission } from '@/features/permission'
import { DictTag } from '@/shared/components/dict-tag'
import { formatDateTime } from '@/shared/lib/format'
import { usePagedQuery } from '@/shared/hooks/use-paged-query'
import { useTableState } from '@/shared/hooks/use-table-state'
import type { SortParams } from '@/shared/types/api'

import { userApi } from '../api'
import { USER_DICT_TYPES, USER_PERMISSIONS, UserStatus } from '../constants'
import type { UserFilters, UserRespDTO } from '../types'
import { UserSearchForm } from './user-search-form'

/**
 * User Management list page.
 *
 * Composition:
 *  - useTableState — in-memory filters/page/sort. Activity keep-alive preserves
 *    across tab switches. F5 resets (no URL sync per Q1=B).
 *  - usePagedQuery — fetches `/page` + builds antd tableProps.
 *  - 3 useMutation: delete, deleteList, updateStatus. Invalidate USER_QUERY_KEY
 *    on success → list refetches.
 *
 * Modals (Create/Edit form, Reset Password) are stubbed in T2.2 with `message.info`
 * toasts. T2.3 wires the form modal; T2.4 wires the reset-password modal.
 *
 * Self-protection (T2.0-D8):
 *  - Cannot delete self (button disabled, checkbox disabled)
 *  - Cannot toggle own status (Switch disabled)
 *  - CAN edit self (own profile), CAN reset own password (workflow valid)
 */

const USER_QUERY_KEY = ['system', 'user'] as const

/** Initial sort: createTime DESC (newest first) per T2.0-D5. */
const INITIAL_SORT: SortParams = { field: 'createTime', order: 'desc' }

export function UserListPage() {
  const { t } = useTranslation()
  const { message, modal } = App.useApp()
  const queryClient = useQueryClient()
  const currentUser = useAppSelector(selectUser)

  const tableState = useTableState<UserFilters>({}, INITIAL_SORT)
  const [selectedRowKeys, setSelectedRowKeys] = useState<number[]>([])

  const { tableProps } = usePagedQuery<UserRespDTO, UserFilters>({
    queryKey: USER_QUERY_KEY,
    queryFn: userApi.page,
    tableState,
  })

  // ===== Mutations =====

  const deleteOneMut = useMutation({
    mutationFn: (id: number) => userApi.delete(id),
    onSuccess: () => {
      message.success(t('systemUser.messages.deleteSuccess'))
      void queryClient.invalidateQueries({ queryKey: USER_QUERY_KEY })
    },
  })

  const deleteBulkMut = useMutation({
    mutationFn: (ids: number[]) => userApi.deleteList(ids),
    onSuccess: (_d, ids) => {
      message.success(t('systemUser.messages.deleteBulkSuccess', { count: ids.length }))
      setSelectedRowKeys([])
      void queryClient.invalidateQueries({ queryKey: USER_QUERY_KEY })
    },
  })

  const updateStatusMut = useMutation({
    mutationFn: (vars: { id: number; status: number }) => userApi.updateStatus(vars),
    onSuccess: () => {
      message.success(t('systemUser.messages.statusUpdateSuccess'))
      void queryClient.invalidateQueries({ queryKey: USER_QUERY_KEY })
    },
  })

  // ===== Helpers =====

  const isSelf = (record: UserRespDTO) => record.id === currentUser?.id

  // ===== Handlers =====

  // Stubbed — replaced in T2.3 / T2.4 with modal opens.
  const handleCreate = () => {
    message.info('Create modal — coming in T2.3')
  }
  const handleEdit = (user: UserRespDTO) => {
    message.info(`Edit modal — coming in T2.3 (user: ${user.username})`)
  }
  const handleResetPassword = (user: UserRespDTO) => {
    message.info(`Reset password — coming in T2.4 (user: ${user.username})`)
  }

  const handleDeleteOne = (user: UserRespDTO) => {
    modal.confirm({
      title: t('systemUser.confirm.deleteOne', { name: user.username }),
      okText: t('common.delete'),
      okType: 'danger',
      cancelText: t('common.cancel'),
      onOk: () => deleteOneMut.mutateAsync(user.id),
    })
  }

  const handleDeleteBulk = () => {
    if (selectedRowKeys.length === 0) return
    modal.confirm({
      title: t('systemUser.confirm.deleteMany', {
        count: selectedRowKeys.length,
      }),
      okText: t('common.delete'),
      okType: 'danger',
      cancelText: t('common.cancel'),
      onOk: () => deleteBulkMut.mutateAsync(selectedRowKeys),
    })
  }

  const handleStatusToggle = (user: UserRespDTO, checked: boolean) => {
    const newStatus = checked ? UserStatus.ENABLED : UserStatus.DISABLED
    updateStatusMut.mutate({ id: user.id, status: newStatus })
  }

  // ===== Columns =====

  const columns: TableColumnsType<UserRespDTO> = [
    {
      title: t('systemUser.table.username'),
      dataIndex: 'username',
      sorter: true,
    },
    {
      title: t('systemUser.table.nickname'),
      dataIndex: 'nickname',
    },
    {
      title: t('systemUser.table.deptName'),
      dataIndex: 'deptName',
      render: (v: string | undefined) => v ?? '-',
    },
    {
      title: t('systemUser.table.mobile'),
      dataIndex: 'mobile',
      render: (v: string | undefined) => v ?? '-',
    },
    {
      title: t('systemUser.table.status'),
      dataIndex: 'status',
      render: (status: number, record) => (
        <HasPermission
          code={USER_PERMISSIONS.update}
          fallback={<DictTag dictType={USER_DICT_TYPES.status} value={status} />}
        >
          <Switch
            checked={status === UserStatus.ENABLED}
            disabled={isSelf(record) || updateStatusMut.isPending}
            onChange={checked => handleStatusToggle(record, checked)}
          />
        </HasPermission>
      ),
    },
    {
      title: t('systemUser.table.createTime'),
      dataIndex: 'createTime',
      sorter: true,
      defaultSortOrder: 'descend',
      render: (v: string) => formatDateTime(v),
    },
    {
      title: t('systemUser.table.actions'),
      key: 'actions',
      width: 240,
      render: (_, record) => (
        <Space size="small">
          <HasPermission code={USER_PERMISSIONS.update}>
            <Button type="link" size="small" onClick={() => handleEdit(record)}>
              {t('systemUser.actions.edit')}
            </Button>
          </HasPermission>
          <HasPermission code={USER_PERMISSIONS.delete}>
            <Button
              type="link"
              size="small"
              danger
              disabled={isSelf(record)}
              onClick={() => handleDeleteOne(record)}
            >
              {t('systemUser.actions.delete')}
            </Button>
          </HasPermission>
          <HasPermission code={USER_PERMISSIONS.updatePassword}>
            <Button type="link" size="small" onClick={() => handleResetPassword(record)}>
              {t('systemUser.actions.resetPassword')}
            </Button>
          </HasPermission>
        </Space>
      ),
    },
  ]

  // ===== Render =====

  return (
    <Card>
      <UserSearchForm
        loading={tableProps.loading}
        onSearch={filters => {
          setSelectedRowKeys([])
          tableState.setFilters(filters)
        }}
        onReset={() => {
          setSelectedRowKeys([])
          tableState.clearFilters()
        }}
      />

      <Space style={{ marginBottom: 16 }}>
        <HasPermission code={USER_PERMISSIONS.create}>
          <Button type="primary" onClick={handleCreate}>
            {t('systemUser.actions.create')}
          </Button>
        </HasPermission>
        <HasPermission code={USER_PERMISSIONS.delete}>
          <Button
            danger
            disabled={selectedRowKeys.length === 0 || deleteBulkMut.isPending}
            onClick={handleDeleteBulk}
          >
            {t('systemUser.actions.deleteSelected', {
              count: selectedRowKeys.length,
            })}
          </Button>
        </HasPermission>
      </Space>

      <Table<UserRespDTO>
        {...tableProps}
        columns={columns}
        rowKey="id"
        rowSelection={{
          selectedRowKeys,
          onChange: keys => setSelectedRowKeys(keys as number[]),
          getCheckboxProps: record => ({
            disabled: isSelf(record),
          }),
        }}
      />
    </Card>
  )
}
```

### Design notes

- **`USER_QUERY_KEY` const at module top**: shared between query + invalidation in mutations. Single source of truth.
- **`updateStatusMut.isPending` disables Switch during request**: prevents race condition where user clicks fast 2x.
- **HasPermission fallback for Switch column**: when user lacks `update` permission, fallback to read-only `<DictTag>`. Column still shows status; just can't edit. Better than empty cell.
- **`defaultSortOrder: 'descend'` on createTime column**: visual default. Must agree with `INITIAL_SORT` constant — both say desc. antd doesn't fire onChange on mount, so no infinite-loop risk.
- **No `sortDirections` override**: antd default `['ascend', 'descend', false]` (3-cycle including "clear") is fine. Q T2.0.5 doesn't specify, accept antd default.
- **`'descend'` vs `'desc'`**: antd uses 'ascend'/'descend' (DOM convention); BE uses 'asc'/'desc'. `usePagedQuery` translates at the boundary (A5 implementation).
- **`message.info` stubs**: chosen over silent buttons so the smoke test confirms wiring. T2.3 replaces both create + edit body. T2.4 replaces reset password body.

## Notes on subtle points

### Why `useState<number[]>` for selectedRowKeys instead of `React.Key[]`

antd Table's `rowSelection.onChange` callback emits `React.Key[]` (string | number). Our `rowKey="id"` is numeric, so coercing to `number[]` at the boundary keeps mutation calls type-safe (`userApi.deleteList(number[])`). The cast `keys as number[]` is justified by the rowKey contract.

### Why `void queryClient.invalidateQueries(...)`

TanStack returns a Promise. We don't await — the UI doesn't need to wait for refetch to complete (loading state shown via `tableProps.loading`). `void` operator suppresses no-floating-promise lint warning.

### Why search submits via `tableState.setFilters` not `setQueryParams`

`useTableState.setFilters(patch)` merges + resets page to 1 + reactive (re-renders triggering re-query). All in one call.

### dept filter "includes child departments"

BE service handles via `deptIdIncludeChildren` lookup in `AdminUserService`. FE just sends `deptId`. Worth knowing for testing: searching by parent dept returns users in all sub-depts.

### Search form ↔ list page contract

`UserSearchForm` doesn't import `useTableState` or know about pagination. It takes `(filters: UserFilters) => void` callback. Decoupled — same form could feed any data sink (e.g., a future export endpoint).

### Strict-mode considerations

- Mutations don't fire twice in strict-mode (only on user interaction).
- `usePagedQuery` runs `useQuery` which dedupes in-flight via key.
- `useTableState` is pure useState — strict-mode double-invoke is idempotent.

### What happens during initial paint

1. `tableState` initializes with `INITIAL_SORT` → queryParams includes sortingFields
2. `usePagedQuery` calls `userApi.page(queryParams)` → returns Promise
3. `tableProps.loading = true`, dataSource empty
4. Table renders with spinner over empty body
5. BE responds → tableProps updates → table renders rows

No flicker because `keepPreviousData` in usePagedQuery (only matters on subsequent fetches; initial load shows spinner).
